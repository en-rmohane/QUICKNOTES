EX. NO: 1 - IMPLEMENTATION OF SYMBOL TABLE

Status: Complete Language: C

AIM:

To write a C program to implement a symbol table that dynamically stores identifier declarations and memory addresses during token analysis.

ALGORITHM:

  1. Start the program.
  2. Read a mathematical/logical expression from the user, terminated by the character $.
  3. Scan the expression character by character.
  4. If a scanned character is an alphabet (identifying a variable name):
    • Check if the next character is an operator (e.g. +, -, *, /, =) or if it's the last character of the expression.
    • If yes, dynamically allocate memory for the symbol identifier using the malloc() function.
    • Insert the identifier character and its allocated virtual memory address into the tracking arrays.
    • Print the identifier, its address, and classification type ("identifier").
  5. After table generation, prompt the user to input a symbol to search.
  6. Search for the symbol in the created lookup table. If found, display its details along with the virtual memory address. Otherwise, notify that it is not found.
  7. Stop the program.

PROGRAM (Standard-Compliant C Code):

symbol_table.c 
#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <string.h>

int main() {
    int i = 0, j = 0, x = 0, n, flag = 0;
    void *p, *add[15];
    char ch, srch, b[15], d[15], c;
    
    printf("Enter expression terminated by $:\n");
    while ((c = getchar()) != '$' && i < 14) {
        b[i] = c;
        i++;
    }
    b[i] = '\0';
    n = i - 1;
    
    printf("\nGiven expression: %s\n", b);
    printf("\nSymbol Table:\n");
    printf("Symbol\tAddress\t\tType\n");
    printf("------------------------------------\n");
    
    while (j <= n) {
        c = b[j];
        if (isalpha((unsigned char)c)) {
            if (j == n) {
                p = malloc(sizeof(char));
                add[x] = p;
                d[x] = c;
                printf("%c\t%p\tidentifier\n", c, p);
                x++;
            } else {
                ch = b[j + 1];
                if (ch == '/' || ch == '+' || ch == '-' || ch == '*' || ch == '=') {
                    p = malloc(sizeof(char));
                    add[x] = p;
                    d[x] = c;
                    printf("%c\t%p\tidentifier\n", c, p);
                    x++;
                }
            }
        }
        j++;
    }
    
    // Clear input buffer
    while ((c = getchar()) != '\n' && c != EOF);
    
    printf("\nEnter the identifier to be searched: ");
    srch = getchar();
    for (i = 0; i < x; i++) {
        if (srch == d[i]) {
            printf("Symbol '%c' found @ Address %p\n", srch, add[i]);
            flag = 1;
            break;
        }
    }
    if (flag == 0) {
        printf("Symbol not found\n");
    }
    
    // Clean up allocated memory
    for (i = 0; i < x; i++) {
        free(add[i]);
    }
    return 0;
}

EX. NO: 2 - DEVELOP A LEXICAL ANALYZER TO RECOGNIZE FEW PATTERNS IN C

Status: Complete Language: C

AIM:

To write a C program to implement a lexical analyzer that processes standard input and classifies character tokens into identifiers, operators, numbers, keywords, and special characters.

ALGORITHM:

  1. Start the program.
  2. Read a sample C program block from input, terminated by EOF (Ctrl+D on Linux/MacOS, Ctrl+Z on Windows). Write this buffer content to a temporary file input.txt.
  3. Open the temporary file for reading. Open separate working files for writing identifiers (identifier.txt) and special characters (specialchar.txt).
  4. Read character-by-character from the source file:
    • If the character is a digit, scan recursively to group digits together into a single numerical token. Store it in a numerical array.
    • If the character is an alphabet, scan recursively to group characters, digits, underscores, and $ symbols to form an identifier token. Write this token to identifier.txt followed by a space.
    • If the character is a newline (\n), increment the line number counter.
    • If it is a space or tab, ignore it.
    • Otherwise, write it to specialchar.txt as a special symbol or operator.
  5. Close all open file streams.
  6. Print all grouped constants/numbers extracted from the input.
  7. Read tokens from identifier.txt. For each token, compare it against a list of reserved C keywords (e.g. for, while, do, int, float, char).
    • If matched, classify it as a **Keyword**.
    • Otherwise, classify it as an **Identifier**.
  8. Read and print the special characters recorded in specialchar.txt.
  9. Print the total number of lines detected in the program block.
  10. Stop the program.

PROGRAM:

lexical_analyzer.c 
#include <stdio.h>
#include <ctype.h>
#include <string.h>
#include <stdlib.h>

void keyword(char str[10]) {
    if (strcmp("for", str) == 0 || strcmp("while", str) == 0 || strcmp("do", str) == 0 ||
        strcmp("int", str) == 0 || strcmp("float", str) == 0 || strcmp("char", str) == 0 ||
        strcmp("double", str) == 0 || strcmp("static", str) == 0 || strcmp("switch", str) == 0 ||
        strcmp("case", str) == 0 || strcmp("void", str) == 0 || strcmp("printf", str) == 0) {
        printf("\n%s is a keyword", str);
    } else {
        printf("\n%s is an identifier", str);
    }
}

int main() {
    FILE *f1, *f2, *f3;
    char c, str[10];
    int num[100], lineno = 1, tokenvalue = 0, i = 0, j = 0, k = 0;
    
    printf("\nEnter the C program (Press Ctrl+D/Ctrl+Z at start of a new line to end):\n");
    f1 = fopen("input.txt", "w");
    if (f1 == NULL) {
        printf("Error opening file for writing!\n");
        return 1;
    }
    while ((c = getchar()) != EOF) {
        putc(c, f1);
    }
    fclose(f1);
    
    f1 = fopen("input.txt", "r");
    f2 = fopen("identifier.txt", "w");
    f3 = fopen("specialchar.txt", "w");
    
    if (f1 == NULL || f2 == NULL || f3 == NULL) {
        printf("Error opening working files!\n");
        return 1;
    }
    
    while ((c = getc(f1)) != EOF) {
        if (isdigit(c)) {
            tokenvalue = c - '0';
            c = getc(f1);
            while (isdigit(c)) {
                tokenvalue = tokenvalue * 10 + (c - '0');
                c = getc(f1);
            }
            num[i++] = tokenvalue;
            ungetc(c, f1);
        } else if (isalpha(c)) {
            putc(c, f2);
            c = getc(f1);
            while (isdigit(c) || isalpha(c) || c == '_' || c == '$') {
                putc(c, f2);
                c = getc(f1);
            }
            putc(' ', f2);
            ungetc(c, f1);
        } else if (c == ' ' || c == '\t') {
            // skip whitespace
        } else if (c == '\n') {
            lineno++;
        } else {
            putc(c, f3);
        }
    }
    
    fclose(f2);
    fclose(f3);
    fclose(f1);
    
    printf("\nThe numbers in the program are:");
    for (j = 0; j < i; j++) {
        printf(" %d", num[j]);
    }
    printf("\n");
    
    f2 = fopen("identifier.txt", "r");
    if (f2 != NULL) {
        printf("\nKeywords and Identifiers classification:");
        while ((c = getc(f2)) != EOF) {
            if (c != ' ') {
                if (k < 9) str[k++] = c;
            } else {
                str[k] = '\0';
                if (k > 0) keyword(str);
                k = 0;
            }
        }
        fclose(f2);
    }
    
    f3 = fopen("specialchar.txt", "r");
    if (f3 != NULL) {
        printf("\n\nSpecial characters are:\n");
        while ((c = getc(f3)) != EOF) {
            printf("%c\t", c);
        }
        printf("\n");
        fclose(f3);
    }
    
    printf("\nTotal number of lines: %d\n", lineno);
    return 0;
}

AIM:

ALGORITHM:

  2. Create a structure block representing a DAG node with label, left pointer, and right pointer attributes.
  3. Request a math expression (containing prior nested parentheses e.g. ((a+b)*(a+b))) from the user.
  4. Locate innermost brackets iteratively and match identifier and operator priorities.
  5. Assign relational parent-child graph connections by storing child node index positions inside left and right pointer entries.
  6. Print the constructed DAG table containing pointer indexes, left nodes, right nodes, and operator/variable labels.

#include <string.h>

struct da {
    int ptr, left, right;
    char label;
} dag[25];

int main() {
    int ptr, j, n = 0, i = 0, x, y;
    char store, input1[50], input[25], var;
    
    for (i = 0; i < 25; i++) {
        dag[i].ptr = i;
        dag[i].left = -1;
        dag[i].right = -1;
        dag[i].label = '\0';
    }
    
    printf("Hint: Provide the expression inside parenthesis stating the priority option\n");
    printf("For Example: a+b*c is given as (a+(b*c))\n");
    printf("\nENTER THE EXPRESSION: ");
    if (scanf("%s", input1) != 1) return 1;
    
    for (i = 0; i < 25; i++) {
        input[i] = '\0';
    }
    
    a:
    for (i = 0; input1[i] != ')' && input1[i] != '\0'; i++);
    if (input1[i] == '\0') goto end_process;
    
    for (j = i; j >= 0 && input1[j] != '('; j--);
    
    for (x = j + 1; x < i; x++) {
        if (isalpha((unsigned char)input1[x])) {
            input[n++] = input1[x];
        } else if (input1[x] != '0') {
            store = input1[x];
            input[n++] = store;
        }
    }
    
    for (x = j; x <= i; x++) {
        input1[x] = '0';
    }
    
    if (input1[0] != '0') goto a;
    
    end_process:
    for (i = 0; i < n; i++) {
        dag[i].label = input[i];
        dag[i].ptr = i;
        if (!isalpha((unsigned char)input[i]) && !isdigit((unsigned char)input[i])) {
            dag[i].right = i - 1;
            ptr = i;
            var = input[i - 1];
            if (isalpha((unsigned char)var)) {
                ptr = ptr - 2;
            } else {
                ptr = i - 1;
                b:
                if (!isalpha((unsigned char)var) && !isdigit((unsigned char)var)) {
                    ptr = dag[ptr].left;
                    var = input[ptr];
                    goto b;
                } else {
                    ptr = ptr - 1;
                }
            }
            dag[i].left = ptr;
        }
    }
    
    printf("\n DAG FOR GIVEN EXPRESSION");
    printf("\n\n PTR \t LEFT PTR \t RIGHT PTR \t LABEL");
    printf("\n-----------------------------------------------------");
    for (i = 0; i < n; i++) {
        printf("\n %d\t\t%d\t\t%d\t\t%c", dag[i].ptr, dag[i].left, dag[i].right, dag[i].label);

}

EX. NO: 9 - IMPLEMENTATION OF SIMPLE CODE OPTIMIZATION TECHNIQUES

Status: Complete Language: C

AIM:

To write a C program to implement basic compiler backend optimization methodologies, including dead code elimination, common subexpression elimination, and duplicate statement elimination.

ALGORITHM:

  1. Start the program.
  2. Input the count of intermediate instructions.
  3. Read three-address code instructions as LHS variable keys and RHS value strings.
  4. **Phase 1: Dead Code Elimination**: Check if the assigned LHS variable is used anywhere in subsequent RHS statements. If it is never queried, eliminate the statement entirely from compilation.
  5. **Phase 2: Common Subexpression Elimination**: Scan RHS strings. If matching expressions are identified, replace the duplicate operations with the original variable reference and update succeeding statements.
  6. **Phase 3: Duplicate Production Elimination**: Compare final generated optimize outputs and remove duplicate definitions.
  7. Print the finalized, optimized intermediate representation.
  8. Stop the program.

PROGRAM (Standard C Code):

code_optimizer.c 
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

struct op {
    char l;
    char r[20];
} op[10], pr[10];

int main() {
    int a, i, k, j, n, z = 0, m, q;
    char *p, *l;
    char temp, t;
    char *tem;
    
    printf("Enter number of values: ");
    if (scanf("%d", &n) != 1) return 1;
    
    for (i = 0; i < n; i++) {
        printf("\tLeft identifier: ");
        scanf(" %c", &op[i].l);
        printf("\tRight expression: ");
        scanf("%s", op[i].r);
    }
    
    printf("\nIntermediate Code:\n");
    for (i = 0; i < n; i++) {
        printf("%c = %s\n", op[i].l, op[i].r);
    }
    
    // Dead Code Elimination
    for (i = 0; i < n - 1; i++) {
        temp = op[i].l;
        for (j = 0; j < n; j++) {
            p = strchr(op[j].r, temp);
            if (p) {
                pr[z].l = op[i].l;
                strcpy(pr[z].r, op[i].r);
                z++;
                break;
            }
        }
    }
    pr[z].l = op[n - 1].l;
    strcpy(pr[z].r, op[n - 1].r);
    z++;
    
    printf("\nAfter Dead Code Elimination:\n");
    for (k = 0; k < z; k++) {
        printf("%c = %s\n", pr[k].l, pr[k].r);
    }
    
    // Common Subexpression Elimination
    for (m = 0; m < z; m++) {
        tem = pr[m].r;
        for (j = m + 1; j < z; j++) {
            p = strstr(tem, pr[j].r);
            if (p) {
                t = pr[j].l;
                pr[j].l = pr[m].l;
                for (i = 0; i < z; i++) {
                    l = strchr(pr[i].r, t);
                    if (l) {
                        a = l - pr[i].r;
                        pr[i].r[a] = pr[m].l;
                    }
                }
            }
        }
    }
    
    // Duplicate statement elimination
    for (i = 0; i < z; i++) {
        for (j = i + 1; j < z; j++) {
            q = strcmp(pr[i].r, pr[j].r);
            if ((pr[i].l == pr[j].l) && !q) {
                pr[i].l = '\0';
                strcpy(pr[i].r, "\0");
            }
        }
    }
    
    printf("\nOptimized Code:\n");
    for (i = 0; i < z; i++) {
        if (pr[i].l != '\0') {
            printf("%c = %s\n", pr[i].l, pr[i].r);
        }
    }
    return 0;
}

EX. NO: 10A - IMPLEMENTATION OF CONTROL FLOW ANALYSIS

Status: Complete Language: C

AIM:

To implement Control Flow Analysis in C by partition intermediate code blocks into Basic Blocks based on program leaders, displaying the control flow topology.

ALGORITHM:

  1. Start the program.
  2. Open the file cdp.txt containing three-address code instructions.
  3. Read and construct a linked list of statements.
  4. Identify basic block **Leaders** based on the following rules:
    • The first statement is a leader.
    • Any statement that is the target of a conditional or unconditional goto (labeled statements like L1:) is a leader.
    • Any statement that immediately follows a conditional or unconditional goto is a leader.
  5. Define basic blocks: A basic block starts at a leader and includes all statements up to, but not including, the next leader.
  6. Map out connections between blocks to display the Control Flow Graph.
  7. Stop the program.

PROGRAM (Standard-Compliant C Code):

control_flow.c 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Listnode {
    char data[50];
    int leader, block, u_goto, c_goto;
    struct Listnode *next;
    char label[10], target[10];
} *temp, *cur, *first = NULL, *last = NULL, *cur1;

void createnode(char code[50]) {
    temp = (struct Listnode *)malloc(sizeof(struct Listnode));
    strcpy(temp->data, code);
    strcpy(temp->label, "");
    strcpy(temp->target, "");
    temp->leader = 0;
    temp->block = 0;
    temp->u_goto = 0;
    temp->c_goto = 0;
    temp->next = NULL;
    
    if (first == NULL) {
        first = temp;
        last = temp;
    } else {
        last->next = temp;
        last = temp;
    }
}

int main() {
    char codeline[50];
    char c, dup[50], target[10];
    char *substring, *token;
    int i = 0, block, block1;
    int j = 0;
    FILE *fpr = fopen("cdp.txt", "r");
    if (!fpr) {
        printf("Error opening cdp.txt file! Create it first.\n");
        return 1;
    }
    
    while ((c = getc(fpr)) != EOF) {
        if (c != '\n') {
            codeline[i] = c;
            i++;
        } else {
            codeline[i] = '\0';
            createnode(codeline);
            i = 0;
        }
    }
    codeline[i] = '\0';
    if(strlen(codeline) > 0) {
        createnode(codeline);
    }
    fclose(fpr);
    
    // Find leaders
    cur = first;
    if (cur != NULL) cur->leader = 1;
    
    while (cur != NULL) {
        substring = strstr(cur->data, "if");
        if (substring == NULL) {
            if (strstr(cur->data, "goto") != NULL) {
                cur->u_goto = 1;
                if (cur->next != NULL) cur->next->leader = 1;
            }
        } else {
            cur->c_goto = 1;
            if (cur->next != NULL) cur->next->leader = 1;
        }
        
        if (strstr(cur->data, ":") != NULL) {
            cur->leader = 1;
        }
        if (strstr(cur->data, "call") != NULL) {
            cur->leader = 1;
        }
        if (strstr(cur->data, "return") != NULL) {
            cur->leader = 1;
            if (cur->next != NULL) cur->next->leader = 1;
        }
        cur = cur->next;
    }
    
    // Extract labels and targets
    cur = first;
    while (cur != NULL) {
        if (cur->u_goto == 1 || cur->c_goto == 1) {
            substring = strstr(cur->data, ":");
            if (substring != NULL) {
                token = strstr(substring, "L");
                if (token != NULL) strcpy(cur->target, token);
            } else {
                substring = strstr(cur->data, "L");
                if (substring != NULL) strcpy(cur->target, substring);
            }
        }
        if (strstr(cur->data, ":") != NULL) {
            strcpy(dup, cur->data);
            token = strtok(dup, ":");
            if (token != NULL) strcpy(cur->label, token);
        }
        cur = cur->next;
    }
    
    // Identify blocks
    cur = first;
    j = 1;
    while (cur != NULL) {
        if (cur->leader == 1) {
            cur->block = j;
            j++;
        } else {
            cur->block = j - 1;
        }
        cur = cur->next;
    }
    
    printf("\n......Basic Blocks......\n");
    cur = first;
    j = 1;
    while (cur != NULL) {
        printf("\nBlock %d:\n", cur->block);
        int current_block = cur->block;
        while(cur != NULL && cur->block == current_block) {
            printf("\t%s\n", cur->data);
            cur = cur->next;
        }
    }
    
    printf("\n.......Control Flow.......\n");
    cur = first;
    while (cur != NULL) {
        if (cur->next != NULL && cur->block != cur->next->block) {
            block = cur->block;
            if (cur->u_goto == 1) {
                strcpy(target, cur->target);
                cur1 = first;
                while (cur1 != NULL) {
                    if (strcmp(cur1->label, target) == 0) {
                        block1 = cur1->block;
                        printf("\tBlock %d ----------> Block %d\n", block, block1);
                    }
                    cur1 = cur1->next;
                }
            } else if (cur->c_goto == 1) {
                strcpy(target, cur->target);
                cur1 = first;
                while (cur1 != NULL) {
                    if (strcmp(cur1->label, target) == 0) {
                        block1 = cur1->block;
                        printf("\tBlock %d ---TRUE---> Block %d / ---FALSE---> Block %d\n", block, block1, block + 1);
                    }
                    cur1 = cur1->next;
                }
            } else if (strstr(cur->data, "return") == NULL) {
                printf("\tBlock %d ----------> Block %d\n", block, block + 1);
            } else {
                printf("\tBlock %d ----------> NULL\n", block);
            }
        }
        cur = cur->next;
    }
    
    // Clean memory
    cur = first;
    while(cur != NULL) {
        struct Listnode *next = cur->next;
        free(cur);
        cur = next;
    }
    return 0;
}
