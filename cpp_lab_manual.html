Experiment No. 1

Class and Object Implementation (Student Record)

AIM / OBJECTIVE
To write a C++ program to create a class named `Student` with data members (Roll Number, Name, Marks in 3 subjects) and member functions to input data, calculate total & percentage, and display student grade card.
SOFTWARE / ENVIRONMENT
G++ / Clang / VS Code / Dev-C++
THEORETICAL BACKGROUND / PREREQUISITES
Object-Oriented Programming models real-world entities into Classes (blueprints) and Objects (instances). Encapsulation bundles data (attributes) and methods (functions) inside a class while enforcing access specifiers (private, public).
STEP-BY-STEP ALGORITHM
  1. Start
  2. Define class Student with private data members: rollNo, name, marks[3], total, percentage.
  3. Define public member functions: inputData(), calculate(), and displayData().
  4. In main(), instantiate an object of Student class.
  5. Call member functions using dot (.) operator.
  6. Display output formatted on console.
  7. Stop
SOURCE CODE (C/C++)
program_1.cpp 
#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

class Student {
private:
    int rollNo;
    string name;
    float marks[3];
    float total;
    float percentage;

public:
    void inputData() {
        cout << "Enter Roll Number: ";
        cin >> rollNo;
        cin.ignore();
        cout << "Enter Student Name: ";
        getline(cin, name);
        cout << "Enter Marks in 3 Subjects (out of 100):\n";
        total = 0;
        for (int i = 0; i < 3; i++) {
            cout << "  Subject " << i + 1 << ": ";
            cin >> marks[i];
            total += marks[i];
        }
        percentage = total / 3.0;
    }

    void displayData() const {
        cout << "\n========================================\n";
        cout << "           STUDENT GRADE REPORT          \n";
        cout << "========================================\n";
        cout << "Roll Number : " << rollNo << endl;
        cout << "Name        : " << name << endl;
        cout << "----------------------------------------\n";
        for (int i = 0; i < 3; i++) {
            cout << "Subject " << i + 1 << "   : " << marks[i] << " / 100" << endl;
        }
        cout << "----------------------------------------\n";
        cout << "Total Marks : " << total << " / 300" << endl;
        cout << fixed << setprecision(2);
        cout << "Percentage  : " << percentage << "%" << endl;
        cout << "Result      : " << (percentage >= 40.0 ? "PASSED" : "FAILED") << endl;
        cout << "========================================\n";
    }
};

int main() {
    Student s1;
    cout << "=== PROGRAM 1: C++ CLASS & OBJECT DEMONSTRATION ===\n";
    s1.inputData();
    s1.displayData();
    return 0;
}

Largest of Three Numbers Using Inline Functions

AIM / OBJECTIVE
To write a C++ program to find the largest of three numbers using an `inline` member function.
SOFTWARE / ENVIRONMENT
G++ / Clang / VS Code
THEORETICAL BACKGROUND / PREREQUISITES
An `inline` function suggests to the compiler to substitute the function body at each call site during compilation, eliminating the overhead of function call stack frames (pushing arguments, jump, return) for small, performance-critical code.
STEP-BY-STEP ALGORITHM
  1. Start
  2. Define inline function findMax(a, b, c) returning the largest integer.
  3. Use nested ternary conditional operators or if-else.
  4. In main(), prompt the user for three numbers.
  5. Call findMax() and display the maximum value.
  6. Stop
SOURCE CODE (C/C++)
program_2.cpp 
#include <iostream>
using namespace std;

// Inline function definition
inline int findMax(int a, int b, int c) {
    return (a > b) ? ((a > c) ? a : c) : ((b > c) ? b : c);
}

int main() {
    int num1, num2, num3;

    cout << "=== PROGRAM 2: INLINE FUNCTION DEMONSTRATION ===\n";
    cout << "Enter three integers: ";
    cin >> num1 >> num2 >> num3;

    int maxVal = findMax(num1, num2, num3);

    cout << "\nAmong [" << num1 << ", " << num2 << ", " << num3 << "]\n";
    cout << "The Largest Number is: " << maxVal << endl;

    return 0;
}

Time Complexity: O(1)
Space Complexity: O(1)
VIVA-VOCE QUESTIONS & ANSWERS

Ans: A function that is expanded in-line at the point of each call by the compiler, avoiding function call overhead.

Swap Private Data Members of Two Classes Using Friend Function

AIM / OBJECTIVE
To write a C++ program to swap the private data values of two distinct classes using a common `friend` function.
SOFTWARE / ENVIRONMENT
G++ / Clang / VS Code
THEORETICAL BACKGROUND / PREREQUISITES
A `friend` function is a non-member function granted access to private and protected members of a class. When swapping data between two classes, pass-by-reference allows in-place modifications.
STEP-BY-STEP ALGORITHM
  1. Start
  2. Forward declare ClassB.
  3. Define ClassA with private member valA and declare friend function swapValues().
  4. Define ClassB with private member valB and declare friend function swapValues().
  5. Implement swapValues(ClassA &a, ClassB &b) to swap using a temp variable.
  6. Instantiate objects, display before swap, call swapValues(), and display after swap.
  7. Stop
SOURCE CODE (C/C++)
program_3.cpp 
#include <iostream>
using namespace std;

class ClassB; // Forward declaration

class ClassA {
private:
    int valA;
public:
    ClassA(int v) : valA(v) {}
    void display() const { cout << "ClassA valA = " << valA << endl; }
    friend void swapValues(ClassA &a, ClassB &b);
};

class ClassB {
private:
    int valB;
public:
    ClassB(int v) : valB(v) {}
    void display() const { cout << "ClassB valB = " << valB << endl; }
    friend void swapValues(ClassA &a, ClassB &b);
};

// Friend function definition
void swapValues(ClassA &a, ClassB &b) {
    int temp = a.valA;
    a.valA = b.valB;
    b.valB = temp;
}

int main() {
    int x, y;
    cout << "=== PROGRAM 3: FRIEND FUNCTION SWAPPING ===\n";
    cout << "Enter value for Class A: ";
    cin >> x;
    cout << "Enter value for Class B: ";
    cin >> y;

    ClassA objA(x);
    ClassB objB(y);

    cout << "\n--- Before Swapping ---\n";
    objA.display();
    objB.display();

    swapValues(objA, objB);

    cout << "\n--- After Swapping (via Friend Function) ---\n";
    objA.display();
    objB.display();

    return 0;
}
SAMPLE INPUT & EXECUTION OUTPUT
Terminal - gcc program_3.cpp && ./a.out
=== PROGRAM 3: FRIEND FUNCTION SWAPPING === Enter value for Class A: 100 Enter value for Class B: 500 --- Before Swapping --- ClassA valA = 100 ClassB valB = 500 --- After Swapping (via Friend Function) --- ClassA valA = 500 ClassB valB = 100
COMPLEXITY ANALYSIS
Time Complexity: O(1)
Space Complexity: O(1)
VIVA-VOCE QUESTIONS & ANSWERS
Q: What is a Friend Function in C++?
Ans: A non-member function defined outside class scope that has full access to private and protected members of the class.
Q: Is friendship mutual or inherited in C++?
Ans: No, friendship is neither symmetric/mutual nor inherited.

Experiment No. 4

Function Overloading (Area of Geometric Shapes)

AIM / OBJECTIVE
To write a C++ program to calculate the area of a Circle, Rectangle, and Triangle using Function Overloading (Compile-Time Polymorphism).
SOFTWARE / ENVIRONMENT
G++ / Clang / VS Code
THEORETICAL BACKGROUND / PREREQUISITES
Function Overloading allows multiple functions in the same scope to share the same name with different parameter signatures (number, order, or data types of parameters). The compiler binds the call based on arguments passed at compile time.
STEP-BY-STEP ALGORITHM
  1. Start
  2. Define overloaded area() functions:
  3. 1. double area(double radius) -> PI * r * r (Circle)
  4. 2. double area(double length, double breadth) -> l * b (Rectangle)
  5. 3. double area(double base, double height, int isTriangle) -> 0.5 * b * h (Triangle)
  6. Prompt user with menu choices.
  7. Invoke corresponding overloaded function and print area.
  8. Stop
SOURCE CODE (C/C++)
program_4.cpp 
#include <iostream>
#include <cmath>
using namespace std;

const double PI = 3.141592653589793;

// Overload 1: Circle
double area(double radius) {
    return PI * radius * radius;
}

// Overload 2: Rectangle
double area(double length, double breadth) {
    return length * breadth;
}

// Overload 3: Triangle
double area(double base, double height, bool isTriangle) {
    if (isTriangle) return 0.5 * base * height;
    return 0;
}

int main() {
    int choice;
    cout << "=== PROGRAM 4: FUNCTION OVERLOADING (AREA CALCULATOR) ===\n";
    cout << "1. Area of Circle\n2. Area of Rectangle\n3. Area of Triangle\nEnter choice (1-3): ";
    cin >> choice;

    switch (choice) {
        case 1: {
            double r;
            cout << "Enter radius of Circle: ";
            cin >> r;
            cout << "=> Area of Circle = " << area(r) << " sq units\n";
            break;
        }
        case 2: {
            double l, b;
            cout << "Enter length and breadth of Rectangle: ";
            cin >> l >> b;
            cout << "=> Area of Rectangle = " << area(l, b) << " sq units\n";
            break;
        }
        case 3: {
            double base, height;
            cout << "Enter base and height of Triangle: ";
            cin >> base >> height;
            cout << "=> Area of Triangle = " << area(base, height, true) << " sq units\n";
            break;
        }
        default:
            cout << "Invalid choice!\n";
    }

    return 0;
}

=== PROGRAM 4: FUNCTION OVERLOADING (AREA CALCULATOR) === 1. Area of Circle 2. Area of Rectangle 3. Area of Triangle Enter choice (1-3): 2 Enter length and breadth of Rectangle: 12.5 8.0 => Area of Rectangle = 100 sq units
COMPLEXITY ANALYSIS
Time Complexity: O(1)
Space Complexity: O(1)
VIVA-VOCE QUESTIONS & ANSWERS
Q: What is Compile-Time Polymorphism?
Ans: Polymorphism resolved during compilation via Function Overloading and Operator Overloading (also called static/early binding).

Ans: No, return type alone is not sufficient to distinguish overloaded functions.

Constructors (Default, Parameterized, Copy) & Destructor

AIM / OBJECTIVE
To write a C++ program demonstrating Default Constructor, Parameterized Constructor, Copy Constructor, and Destructor lifecycle of objects.
SOFTWARE / ENVIRONMENT
G++ / Clang / VS Code
THEORETICAL BACKGROUND / PREREQUISITES
A Constructor is a special member function having the same name as the class invoked automatically upon object creation to initialize state. A Destructor (~ClassName) is invoked automatically when an object goes out of scope to reclaim resources.
STEP-BY-STEP ALGORITHM
  1. Start
  2. Define class Account with account number and balance.
  3. Implement Default Constructor (initialize to default 0).
  4. Implement Parameterized Constructor (initialize with arguments).
  5. Implement Copy Constructor (deep copy another object's state).
  6. Implement Destructor to print deallocation message.
  7. Create objects in main() and observe invocation orders.
  8. Stop
SOURCE CODE (C/C++)
program_5.cpp 
#include <iostream>
#include <string>
using namespace std;

class Account {
private:
    int accNo;
    string holderName;
    double balance;

public:
    // 1. Default Constructor
    Account() {
        accNo = 0;
        holderName = "Unknown";
        balance = 0.0;
        cout << "[*] Default Constructor Called (Acc: " << accNo << ")\n";
    }

    // 2. Parameterized Constructor
    Account(int no, string name, double bal) {
        accNo = no;
        holderName = name;
        balance = bal;
        cout << "[*] Parameterized Constructor Called for " << holderName << " (Acc: " << accNo << ")\n";
    }

    // 3. Copy Constructor
    Account(const Account &other) {
        accNo = other.accNo + 1; // new cloned account
        holderName = other.holderName + " (Joint)";
        balance = other.balance;
        cout << "[*] Copy Constructor Called (Copied from Acc " << other.accNo << " to Acc " << accNo << ")\n";
    }

    // Member display function
    void display() const {
        cout << "   [Account " << accNo << "] " << holderName << " | Balance: $" << balance << endl;
    }

    // Destructor
    ~Account() {
        cout << "[~] Destructor Called: Reclaiming Account " << accNo << " (" << holderName << ")\n";
    }
};

int main() {
    cout << "=== PROGRAM 5: CONSTRUCTORS & DESTRUCTOR LIFECYCLE ===\n\n";
    {
        cout << "--- 1. Creating default object ---\n";
        Account acc1;
        acc1.display();

        cout << "\n--- 2. Creating parameterized object ---\n";
        Account acc2(1001, "Amit Verma", 75000.50);
        acc2.display();

        cout << "\n--- 3. Creating object using Copy Constructor ---\n";
        Account acc3 = acc2; // Copy constructor invocation
        acc3.display();

        cout << "\n--- Leaving Inner Scope (Destructors will trigger in reverse order) ---\n";
    }
    cout << "\nAll accounts destroyed successfully.\n";
    return 0;
}

Creating parameterized object --- [*] Parameterized Constructor Called for Amit Verma (Acc: 1001) [Account 1001] Amit Verma | Balance: $75000.5 --- 3. VIVA-VOCE QUESTIONS & ANSWERS
Q: Why is the parameter to a copy constructor passed by reference (const ClassName &obj)?
Ans: If passed by value, passing the argument would itself call the copy constructor, causing infinite recursive calls and stack overflow.
Q: Can a destructor be overloaded in C++?
Ans: No, a class has exactly one destructor with no parameters and no return type.

Constructor Overloading and Dynamic Memory Allocation

AIM / OBJECTIVE
To write a C++ program demonstrating Constructor Overloading and Dynamic Array allocation using `new` and `delete` operators.
SOFTWARE / ENVIRONMENT
G++ / Clang / VS Code
THEORETICAL BACKGROUND / PREREQUISITES
Constructor Overloading occurs when a class has multiple constructors with different argument lists. Dynamic memory allocation allocates heap memory at runtime using `new[]` and releases it using `delete[]` in destructor preventing memory leaks.
STEP-BY-STEP ALGORITHM
  1. Start
  2. Define class DynamicVector with pointer int *arr and size n.
  3. Constructor 1 (DynamicVector()): allocates default size 3.
  4. Constructor 2 (DynamicVector(int size)): dynamically allocates array of custom size.
  5. Constructor 3 (DynamicVector(int size, int initVal)): allocates and fills with initVal.
  6. Destructor ~DynamicVector(): executes delete[] arr.
  7. In main(), test all constructors and display contents.
  8. Stop
SOURCE CODE (C/C++)
program_6.cpp 
#include <iostream>
using namespace std;

class DynamicVector {
private:
    int* arr;
    int size;

public:
    // Overloaded Constructor 1: Default
    DynamicVector() {
        size = 3;
        arr = new int[size]{10, 20, 30};
        cout << "[*] Default Constructor: Allocated array of size " << size << endl;
    }

    // Overloaded Constructor 2: Parameterized (Size only)
    DynamicVector(int s) {
        size = s;
        arr = new int[size];
        for (int i = 0; i < size; i++) arr[i] = (i + 1) * 100;
        cout << "[*] Parameterized Constructor: Allocated custom size " << size << endl;
    }

    // Overloaded Constructor 3: Parameterized (Size and default fill value)
    DynamicVector(int s, int fillVal) {
        size = s;
        arr = new int[size];
        for (int i = 0; i < size; i++) arr[i] = fillVal;
        cout << "[*] Parameterized Constructor: Allocated size " << size << " filled with " << fillVal << endl;
    }

    void display() const {
        cout << "Vector Elements: [ ";
        for (int i = 0; i < size; i++) {
            cout << arr[i] << " ";
        }
        cout << "]\n";
    }

    // Destructor to free heap memory
    ~DynamicVector() {
        delete[] arr;
        cout << "[~] Destructor: Dynamically allocated memory freed.\n";
    }
};

int main() {
    cout << "=== PROGRAM 6: CONSTRUCTOR OVERLOADING & DYNAMIC MEMORY ===\n\n";
    
    cout << "1. Object V1 (Default):\n";
    DynamicVector v1;
    v1.display();

    cout << "\n2. Object V2 (Custom size 4):\n";
    DynamicVector v2(4);
    v2.display();

    cout << "\n3. Object V3 (Custom size 5 filled with 77):\n";
    DynamicVector v3(5, 77);
    v3.display();

    cout << "\n--- Scope Exit ---\n";
    return 0;
}

Experiment No. 7

Unary Operator Overloading (Overload - and ++ Operators)

AIM / OBJECTIVE
To write a C++ program to overload Unary Minus (`-`) and Prefix/Postfix Increment (`++`) operators.
SOFTWARE / ENVIRONMENT
G++ / Clang / VS Code
THEORETICAL BACKGROUND / PREREQUISITES
Operator Overloading allows C++ operators to be given special user-defined meanings when applied to user-defined class objects. Unary operators operate on a single operand (the calling object `*this`).
STEP-BY-STEP ALGORITHM
  1. Start
  2. Define class Point with x and y coordinates.
  3. Overload Unary Minus: Point operator-() const -> returns Point(-x, -y).
  4. Overload Prefix ++: Point& operator++() -> increments ++x, ++y, returns *this.
  5. Overload Postfix ++: Point operator++(int) -> saves old state, increments, returns old state.
  6. In main(), create Point and apply unary operators.
  7. Stop
SOURCE CODE (C/C++)
program_7.cpp 
#include <iostream>
using namespace std;

class Point {
private:
    int x, y;

public:
    Point(int x = 0, int y = 0) : x(x), y(y) {}

    void display() const {
        cout << "(" << x << ", " << y << ")" << endl;
    }

    // 1. Overloading Unary Minus (-)
    Point operator-() const {
        return Point(-x, -y);
    }

    // 2. Overloading Prefix Increment (++p)
    Point& operator++() {
        ++x;
        ++y;
        return *this;
    }

    // 3. Overloading Postfix Increment (p++)
    Point operator++(int) {
        Point temp = *this; // save old state
        x++;
        y++;
        return temp;        // return old state
    }
};

int main() {
    Point p1(15, -25);

    cout << "=== PROGRAM 7: UNARY OPERATOR OVERLOADING ===\n";
    cout << "Original Point P1: ";
    p1.display();

    // Applying Unary Minus
    Point pNeg = -p1;
    cout << "Negated Point (-P1): ";
    pNeg.display();

    // Applying Prefix ++
    cout << "\nApplying Prefix ++ (++P1): ";
    (++p1).display();

    // Applying Postfix ++
    cout << "Applying Postfix ++ (P1++): ";
    (p1++).display();
    cout << "Point P1 after Postfix Increment: ";
    p1.display();

    return 0;
}
SAMPLE INPUT & EXECUTION OUTPUT
Terminal - gcc program_7.cpp && ./a.out
=== PROGRAM 7: UNARY OPERATOR OVERLOADING === Original Point P1: (15, -25) Negated Point (-P1): (-15, 25) Applying Prefix ++ (++P1): (16, -24) Applying Postfix ++ (P1++): (16, -24) Point P1 after Postfix Increment: (17, -23)
COMPLEXITY ANALYSIS
Time Complexity: O(1)
Space Complexity: O(1)
VIVA-VOCE QUESTIONS & ANSWERS
Q: How does compiler differentiate between prefix and postfix operator++ in overloading?
Ans: Postfix takes a dummy `int` argument in its signature: operator++(int).
Q: Which operators cannot be overloaded in C++?
Ans: Scope resolution (::), Member selection (.), Member selection via pointer (.*), Ternary (?:), and sizeof.

AIM / OBJECTIVE
To write a C++ program to overload the Binary Plus (`+`) operator to add two Complex numbers.
SOFTWARE / ENVIRONMENT

using namespace std;

class Complex {
private:
    float real;
    float imag;

public:
    Complex(float r = 0.0, float i = 0.0) : real(r), imag(i) {}

    void input() {
        cout << "Enter Real and Imaginary parts: ";
        cin >> real >> imag;
    }

    void display() const {
        if (imag >= 0)
            cout << real << " + " << imag << "i" << endl;
        else
            cout << real << " - " << -imag << "i" << endl;
    }

    // Overloading Binary + Operator
    Complex operator+(const Complex &c) const {
        Complex temp;
        temp.real = this->real + c.real;
        temp.imag = this->imag + c.imag;
        return temp;
    }
};

    // Adding using overloaded + operator
    c3 = c1 + c2; // Equivalent to c1.operator+(c2)

    cout << "\n--- Results ---\n";
    cout << "C1 = "; c1.display();

// Abstract Base Class
class Shape {
public:
    // Pure Virtual Function
    virtual void calculateArea() = 0;
    virtual void displayInfo() const = 0;

    // Virtual Destructor for safe polymorphic destruction
    virtual ~Shape() {
        cout << "[~] Shape Base Destructor\n";
    }
};

class Circle : public Shape {
private:
    double radius;
public:
    Circle(double r) : radius(r) {}
    void calculateArea() override {
        cout << "Circle Area: " << (3.14159 * radius * radius) << " sq units\n";

    ~Circle() { cout << "[~] Circle Destructor\n"; }

private:
    double length, width;

        cout << "Rectangle Area: " << (length * width) << " sq units\n";

    void displayInfo() const override {

};

#include <string>

// 1. Generic Function Template
template <typename T>
void customSwap(T &a, T &b) {
    T temp = a;
    a = b;
    b = temp;
}

// 2. Generic Class Template (Stack)
template <typename T>
class GenericStack {
private:
    T arr[10];
    int top;
public:
    GenericStack() : top(-1) {}

    void push(T val) {
        if (top == 9) {
            cout << "[!] Stack Overflow\n";
        } else {
            arr[++top] = val;
        }
    }

    T pop() {
        if (top == -1) {
            cout << "[!] Stack Underflow\n";
            return T();
        }
        return arr[top--];
    }

    void display() const {
        cout << "[Stack]: ";
        for (int i = 0; i <= top; i++) {
            cout << arr[i] << " ";
        }
        cout << endl;
    }
};
